# SOVEREIGN GATEWAY SERVER RULES
# 1. APACHE (.htaccess) İçin Kurallar
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /

  # Mevcut dosya veya klasörlere dokunma (CSS, JS, resimler vb.)
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d

  # Eğer URL /en, /ru gibi bir sanal dil koduyla başlıyorsa
  # O kodu yoksay ve geri kalan ana dizindeki dosyayı çalıştır.
  # Örn: /en/ritueller.html -> /ritueller.html
  RewriteRule ^[a-z]{2}/(.*)$ /$1 [L,QSA]
</IfModule>
